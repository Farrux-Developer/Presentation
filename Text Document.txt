    <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/gsap.min.js"></script>
    <script src="https://unpkg.com/split-type"></script>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r134/three.min.js"></script>

    <script>
        // --- 0. Intro Animation (Opening Eyes) ---
        window.addEventListener('load', () => {
            const tlIntro = gsap.timeline({
                onComplete: () => {
                    // Только после открытия глаз запускаем анимацию текста
                    animateText(0);
                    // Удаляем шторки из DOM для оптимизации
                    document.querySelectorAll('.eyelid').forEach(el => el.remove());
                }
            });

            // Небольшая задержка для саспенса, затем шторки разъезжаются
            tlIntro.to('.eyelid-top', { yPercent: -100, duration: 1.5, ease: "power4.inOut", delay: 0.5 }, 0)
                   .to('.eyelid-bottom', { yPercent: 100, duration: 1.5, ease: "power4.inOut", delay: 0.5 }, 0)
                   // Карточка слегка увеличивается из темноты
                   .fromTo('.glass-card.ui-element', 
                        { scale: 0.95, opacity: 0 }, 
                        { scale: 1, opacity: 1, duration: 1.5, ease: "power3.out" }, 0.8)
                   // Проявление интерфейса
                   .to('.ui-element:not(.glass-card)', { opacity: 1, duration: 1, ease: "power2.out" }, 1);
        });

        // --- 1. Dynamic Spotlight & Micro-Parallax ---
        const isMobile = window.innerWidth <= 768;
        let mouseX = window.innerWidth / 2;
        let mouseY = window.innerHeight / 2;

        window.addEventListener('mousemove', (e) => {
            mouseX = e.clientX; mouseY = e.clientY;
        });

        document.querySelectorAll('.glass-card').forEach(card => {
            if(!isMobile) {
                card.addEventListener('mousemove', (e) => {
                    const rect = card.getBoundingClientRect();
                    const x = e.clientX - rect.left;
                    const y = e.clientY - rect.top;
                    
                    card.style.setProperty('--mouse-x', `${x}px`);
                    card.style.setProperty('--mouse-y', `${y}px`);

                    const centerX = rect.width / 2;
                    const centerY = rect.height / 2;
                    const moveX = (x - centerX) * 0.03;
                    const moveY = (y - centerY) * 0.03;
                    
                    gsap.to(card.querySelector('.card-content'), {
                        x: moveX, y: moveY, duration: 0.5, ease: "power2.out"
                    });
                });

                card.addEventListener('mouseleave', () => {
                    gsap.to(card.querySelector('.card-content'), {
                        x: 0, y: 0, duration: 0.5, ease: "power2.out"
                    });
                });
            }
        });

        // --- 2. SplitType & Typography Animation ---
        const splitTextElements = new SplitType('.split', { types: 'chars, words' });
        
        function animateText(slideIndex) {
            const slide = slides[slideIndex];
            const chars = slide.querySelectorAll('.char');
            const p = slide.querySelectorAll('.fade-text');
            const tag = slide.querySelectorAll('.tag');

            gsap.fromTo(chars, 
                { y: 100, opacity: 0 }, 
                { y: 0, opacity: 1, duration: 1, stagger: 0.02, ease: "power4.out", delay: 0.1 }
            );
            gsap.fromTo([tag, p],
                { y: 20, opacity: 0 },
                { y: 0, opacity: 1, duration: 1.2, ease: "power3.out", delay: 0.5, stagger: 0.2 }
            );
        }

        // --- 3. WebGL Background Setup (Optimized) ---
        const canvas = document.getElementById('webgl-canvas');
        const scene = new THREE.Scene();
        const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
        
        const renderer = new THREE.WebGLRenderer({ 
            canvas, 
            antialias: false,
            alpha: false,
            powerPreference: "high-performance" 
        });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 1.5)); 

        const loader = new THREE.TextureLoader();
        const images = ["./Img/image.png", "./Img/ocean.jpg", "./Img/bubble_gum.jpg"];
        const textures = images.map(img => loader.load(img));

        const vertexShader = `
            varying vec2 vUv;
            void main() { vUv = uv; gl_Position = vec4(position, 1.0); }
        `;
        const fragmentShader = `
            uniform sampler2D texCurrent;
            uniform sampler2D texNext;
            uniform float progress;
            uniform float time;
            uniform float aspect;
            varying vec2 vUv;

            void main() {
                vec2 uv = vUv;
                vec2 centered = uv - 0.5;
                centered.y /= aspect;
                
                float wave = sin(uv.y * 15.0 + time * 2.0) * 0.03 * sin(progress * 3.1415);
                vec2 distortedUv = uv + vec2(wave, 0.0);
                
                vec2 uv1 = (distortedUv - 0.5) * (1.0 - progress * 0.1) + 0.5;
                vec2 uv2 = (distortedUv - 0.5) * (1.0 + (1.0 - progress) * 0.1) + 0.5;

                vec4 color1 = texture2D(texCurrent, uv1);
                vec4 color2 = texture2D(texNext, uv2);

                gl_FragColor = mix(color1, color2, progress);
            }
        `;

        const uniforms = {
            texCurrent: { value: textures[0] },
            texNext: { value: textures[1] },
            progress: { value: 0.0 },
            time: { value: 0.0 },
            aspect: { value: window.innerWidth / window.innerHeight }
        };

        const bgMaterial = new THREE.ShaderMaterial({
            vertexShader, fragmentShader, uniforms, depthWrite: false
        });
        const bgPlane = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), bgMaterial);
        scene.add(bgPlane);

        const pCamera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 100);
        pCamera.position.z = 5;
        
        const pGeo = new THREE.BufferGeometry();
        const pCount = isMobile ? 300 : 600; 
        const pPos = new Float32Array(pCount * 3);
        for(let i=0; i<pCount*3; i++) pPos[i] = (Math.random() - 0.5) * 15;
        pGeo.setAttribute('position', new THREE.BufferAttribute(pPos, 3));
        
        const pMat = new THREE.PointsMaterial({ 
            size: 0.02, color: 0x88ccff, transparent: true, opacity: 0.0, blending: THREE.AdditiveBlending 
        });
        const pMesh = new THREE.Points(pGeo, pMat);
        scene.add(pMesh);

        const clock = new THREE.Clock();
        function render() {
            requestAnimationFrame(render);
            const time = clock.getElapsedTime();
            uniforms.time.value = time;
            
            if(pMat.opacity > 0) {
                pMesh.rotation.y = time * 0.02;
                pMesh.rotation.x = time * 0.01;
                pCamera.position.x += ((mouseX / window.innerWidth - 0.5) * 2 - pCamera.position.x) * 0.05;
                pCamera.position.y += (-(mouseY / window.innerHeight - 0.5) * 2 - pCamera.position.y) * 0.05;
            }

            renderer.render(scene, camera);
            if(pMat.opacity > 0) {
                renderer.autoClear = false;
                renderer.render(scene, pCamera);
                renderer.autoClear = true;
            }
        }
        render();

        // --- 4. Main Scroll & Swipe Logic ---
        const slides = document.querySelectorAll('.slide');
        const dots = document.querySelectorAll('.dot');
        let currentSlide = 0;
        let isAnimating = false;

        window.addEventListener('wheel', (e) => {
            if(isAnimating) return;
            if(e.deltaY > 50 && currentSlide < slides.length - 1) changeSlide(currentSlide + 1);
            else if(e.deltaY < -50 && currentSlide > 0) changeSlide(currentSlide - 1);
        });

        let touchStartY = 0;
        window.addEventListener('touchstart', e => {
            touchStartY = e.touches[0].clientY;
        }, { passive: true });

        window.addEventListener('touchend', e => {
            if(isAnimating) return;
            const touchEndY = e.changedTouches[0].clientY;
            const diff = touchStartY - touchEndY;
            
            if(diff > 50 && currentSlide < slides.length - 1) changeSlide(currentSlide + 1);
            else if(diff < -50 && currentSlide > 0) changeSlide(currentSlide - 1);
        });

        function changeSlide(nextIndex) {
            isAnimating = true;
            const direction = nextIndex > currentSlide ? 1 : -1;
            
            dots[currentSlide].classList.remove('active');
            dots[nextIndex].classList.add('active');

            uniforms.texCurrent.value = textures[currentSlide];
            uniforms.texNext.value = textures[nextIndex];
            uniforms.progress.value = 0.0;

            const tl = gsap.timeline({
                onComplete: () => {
                    slides[currentSlide].classList.remove('active');
                    currentSlide = nextIndex;
                    slides[currentSlide].classList.add('active');
                    animateText(currentSlide);
                    isAnimating = false;
                }
            });

            tl.to(slides[currentSlide].querySelector('.glass-card'), { opacity: 0, y: -20 * direction, duration: 0.6, ease: "power2.in" }, 0);
            tl.to(uniforms.progress, { value: 1.0, duration: 1.5, ease: "power3.inOut" }, 0);
            
            if(nextIndex === 1) {
                gsap.to(pMat, { opacity: 0.6, duration: 2, ease: "power2.out", delay: 0.5 });
            } else {
                gsap.to(pMat, { opacity: 0, duration: 1, ease: "power2.in" });
            }

            tl.set(slides[nextIndex], { className: "slide active" }, 0.6);
            tl.fromTo(slides[nextIndex].querySelector('.glass-card'), 
                { opacity: 0, y: 50 * direction }, 
                { opacity: 1, y: 0, duration: 1, ease: "power3.out" }, 1.0);
        }

        // --- 5. Hold to Explore Interaction ---
        let holdTimer;
        let isExploring = false;

        document.querySelectorAll('.glass-card').forEach(card => {
            const startEvent = isMobile ? 'touchstart' : 'mousedown';
            const endEvent = isMobile ? 'touchend' : 'mouseup';

            card.addEventListener(startEvent, () => {
                if(isAnimating) return;
                holdTimer = setTimeout(() => {
                    isExploring = true;
                    gsap.to('.glass-card, .scroll-indicator, .hint', { opacity: 0, duration: 1, ease: "power2.inOut" });
                    if(currentSlide === 1) gsap.to(pMat, { opacity: 0, duration: 1 });
                }, 600);
            }, { passive: true });

            card.addEventListener(endEvent, () => {
                clearTimeout(holdTimer);
                if(isExploring) {
                    isExploring = false;
                    gsap.to('.glass-card, .scroll-indicator, .hint', { opacity: 1, duration: 1, ease: "power2.inOut" });
                    if(currentSlide === 1) gsap.to(pMat, { opacity: 0.6, duration: 1 });
                }
            }, { passive: true });
        });

        // --- 6. Защита от лагов при ресайзе (Debounce) ---
        let resizeTimeout;
        window.addEventListener('resize', () => {
            clearTimeout(resizeTimeout);
            resizeTimeout = setTimeout(() => {
                const width = window.innerWidth;
                const height = window.innerHeight;
                
                renderer.setSize(width, height);
                uniforms.aspect.value = width / height;
                
                pCamera.aspect = width / height;
                pCamera.updateProjectionMatrix();
            }, 150);
        });

    </script>
